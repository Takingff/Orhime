# ============================================================
# YOUTUBE SEARCH
# ============================================================

async def youtube_search(query):

    loop = asyncio.get_running_loop()

    def extract():

        if query.startswith("http://") or query.startswith("https://"):
            search_query = query
        else:
            search_query = f"ytsearch1:{query}"

        return ytdl.extract_info(
            search_query,
            download=False
        )

    try:

        data = await asyncio.wait_for(
            loop.run_in_executor(
                None,
                extract
            ),
            timeout=30
        )

        if not data:
            return None

        # Search result
        if "entries" in data:

            entries = data.get("entries")

            if not entries:
                return None

            return entries[0]

        # Direct URL
        return data

    except asyncio.TimeoutError:

        print("YouTube search timed out.")
        return None

    except Exception as e:

        print(f"YouTube error: {repr(e)}")
        return None


# ============================================================
# CREATE AUDIO SOURCE
# ============================================================

def create_audio_source(info):

    audio_url = info.get("url")

    if not audio_url:
        raise RuntimeError(
            "YouTube did not return an audio URL."
        )

    # IMPORTANT:
    #
    # FFmpegPCMAudio decodes the incoming stream into PCM.
    #
    # This prevents:
    #
    # AudioSource must not be Opus encoded
    #
    source = discord.FFmpegPCMAudio(
        audio_url,
        executable=FFMPEG_PATH,
        before_options=FFMPEG_BEFORE_OPTIONS,
        options=FFMPEG_OPTIONS
    )

    return source


# ============================================================
# PLAY NEXT SONG
# ============================================================

async def play_next(guild):

    guild_id = guild.id

    queue = get_queue(guild_id)

    voice_client = guild.voice_client

    if not voice_client:
        return

    if not queue:

        print(
            f"[{guild.name}] Queue is empty."
        )

        return

    song = queue.pop(0)

    title = song["title"]
    info = song["info"]

    try:

        source = create_audio_source(info)

    except Exception as e:

        print(
            f"Audio source creation error: {repr(e)}"
        )

        await play_next(guild)

        return

    def after(error):

        if error:
            print(
                f"Playback error: {repr(error)}"
            )

        # Schedule next song safely
        asyncio.run_coroutine_threadsafe(
            play_next(guild),
            bot.loop
        )

    try:

        voice_client.play(
            source,
            after=after
        )

        print(
            f"[{guild.name}] Now playing: {title}"
        )

    except Exception as e:

        print(
            f"Voice playback error: {repr(e)}"
        )

        await play_next(guild)

# PLAY COMMAND
# ============================================================

@bot.command(
    name="play",
    aliases=["p"]
)
async def play(ctx, *, query=None):

    if not query:

        await ctx.send(
            f"❌ Use `{PREFIX}play song name`"
        )

        return

    # --------------------------------------------------------
    # Check voice channel
    # --------------------------------------------------------

    if not ctx.author.voice:

        await ctx.send(
            "❌ You need to join a voice channel first."
        )

        return

    user_channel = ctx.author.voice.channel

    # --------------------------------------------------------
    # Connect to voice
    # --------------------------------------------------------

    voice_client = ctx.guild.voice_client

    try:

        if voice_client is None:

            voice_client = await user_channel.connect(
                reconnect=True
            )

        elif voice_client.channel != user_channel:

            await voice_client.move_to(
                user_channel
            )

    except Exception as e:

        print(
            f"Voice connection error: {repr(e)}"
        )

        await ctx.send(
            "❌ I couldn't connect to the voice channel."
        )

        return

    # --------------------------------------------------------
    # Searching
    # --------------------------------------------------------

    message = await ctx.send(
        f"🔎 Searching YouTube for **{query}**..."
    )

    print(
        f"Searching YouTube: {query}"
    )

    info = await youtube_search(query)

    if not info:

        await message.edit(
            content=(
                f"❌ I couldn't find **{query}**.\n"
                "Try another song name or YouTube URL."
            )
        )

        return

    # --------------------------------------------------------
    # Song information
    # --------------------------------------------------------

    title = info.get(
        "title",
        "Unknown Song"
    )

    webpage_url = info.get(
        "webpage_url",
        query
    )

    duration = info.get(
        "duration",
        0
    )

    # --------------------------------------------------------
    # Add to queue
    # --------------------------------------------------------

    queue = get_queue(
        ctx.guild.id
    )

    was_playing = voice_client.is_playing()

    song = {
        "title": title,
        "url": webpage_url,
        "duration": duration,
        "info": info
    }

    queue.append(song)

    # --------------------------------------------------------
    # Start immediately if nothing playing
    # --------------------------------------------------------

    if not was_playing:

        await message.edit(
            content=f"⏳ Loading **{title}**..."
        )

        await play_next(
            ctx.guild
        )

        await asyncio.sleep(1)

        if voice_client.is_playing():

            await message.edit(
                content=f"🎵 **Now playing:** {title}"
            )

        else:

            await message.edit(
                content=(
                    f"❌ Couldn't play **{title}**."
                )
            )

    else:

        position = len(queue)

        await message.edit(
            content=(
                f"✅ Added to queue:\n"
                f"**{title}**\n"
                f"📋 Position: `{position}`"
            )
        )

@bot.command(name="stop")

    voice_client = ctx.guild.voice_client

    if not voice_client:

        await ctx.send(
            "❌ I'm not connected to voice."
        )

        return

    await ctx.send(
        "⏹️ Stopped playback and cleared the queue."
    )

@bot.command(
    name="queue",
    aliases=["q"]
)
async def queue_command(ctx):

    queue = get_queue(
        ctx.guild.id
    )

    voice_client = ctx.guild.voice_client

    if not queue:

        if voice_client and voice_client.is_playing():

            await ctx.send(
                "🎵 Currently playing.\n"
                "📋 Queue is empty."
            )

        else:

            await ctx.send(
                "📋 The queue is empty."
            )

        return

    embed = discord.Embed(
        title="📋 Music Queue",
        color=discord.Color.blurple()
    )

    for index, song in enumerate(
        queue[:10],
        start=1
    ):

        embed.add_field(
            name=f"{index}. {song['title']}",
            value="Queued",
            inline=False
        )

    if len(queue) > 10:

        embed.set_footer(
            text=(
                f"{len(queue) - 10} more songs..."
            )
        )

    await ctx.send(
        embed=embed
    )

@bot.event
async def on_command_error(ctx, error):

            f"❌ Missing argument. Try `{PREFIX}help`."
